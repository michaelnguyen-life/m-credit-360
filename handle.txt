function handleFileSelect(e) {
      const files = e.target.files;
      if (!files.length) return;
      currentUploadedFileName = files[0].name || "BCTC_Uploaded.pdf";

      const sourceTag = document.getElementById('eb-source-tag');
      if (sourceTag) {
        sourceTag.textContent = "Nguồn: " + currentUploadedFileName + " (Đang bóc tách AI)";
      }

      const box = document.getElementById('ai-processing-box');
      if (box) box.classList.remove('hidden');

      const modal = document.getElementById('ai-processing-modal');
      const modalFilename = document.getElementById('ai-modal-filename');
      const modalBar = document.getElementById('ai-modal-bar');
      const modalPct = document.getElementById('ai-modal-pct');
      const modalStage = document.getElementById('ai-modal-stage');

      if (modalFilename) modalFilename.textContent = currentUploadedFileName;
      if (modal) modal.classList.remove('hidden');
      if (typeof lucide !== 'undefined') lucide.createIcons();

      const stages = [
        { pct: 15,  stage: "Giai đoạn 1/6: Khởi tạo pipeline AI...",         step: 1 },
        { pct: 30,  stage: "Giai đoạn 2/6: OCR / đọc tài liệu BCTC scan...",  step: 2 },
        { pct: 50,  stage: "Giai đoạn 3/6: Phân loại hồ sơ & đối chiếu...",   step: 3 },
        { pct: 70,  stage: "Giai đoạn 4/6: Trích xuất chỉ tiêu tài chính MSB...", step: 4 },
        { pct: 85,  stage: "Giai đoạn 5/6: Validate & cross-check Rule 5D...", step: 5 },
        { pct: 100, stage: "Giai đoạn 6/6: Hoàn tất bóc tách dữ liệu!",       step: 6 },
      ];

      let idx = 0;
      const interval = setInterval(() => {
        if (idx >= stages.length) {
          clearInterval(interval);
          setTimeout(() => {
            if (modal) modal.classList.add('hidden');
            closeUploadDrawer();
            if (sourceTag) sourceTag.textContent = "Nguồn: " + currentUploadedFileName + " (Đã bóc tách AI)";
            
            let detectedType = 'alpha';
            const lowerName = currentUploadedFileName.toLowerCase();
            if (lowerName.includes('khang') || lowerName.includes('thinh')) detectedType = 'eb_good_1';
            else if (lowerName.includes('vinamilk') || lowerName.includes('vnm')) detectedType = 'eb_good_2';
            else if (lowerName.includes('flc')) detectedType = 'eb_bad_1';
            else if (lowerName.includes('tanhoangminh') || lowerName.includes('thm')) detectedType = 'eb_bad_2';
            
            openExtractionReview(detectedType);
          }, 500);
          return;
        }
        const s = stages[idx];
        if (modalBar) modalBar.style.width = s.pct + '%';
        if (modalPct) modalPct.textContent = s.pct + '%';
        if (modalStage) modalStage.textContent = s.stage;

        const bar = document.getElementById('ai-stage-bar');
        const pctText = document.getElementById('ai-stage-pct');
        const stageText = document.getElementById('ai-stage-text');
        if (bar) bar.style.width = s.pct + '%';
        if (pctText) pctText.textContent = s.pct + '%';
        if (stageText) stageText.textContent = s.stage;

        for (let i = 1; i <= 6; i++) {
          const stepEl = document.getElementById('ai-step-' + i);
          if (!stepEl) continue;
          if (i < s.step) {
            stepEl.className = 'flex items-center gap-2 text-[#00A86B]';
            stepEl.innerHTML = '<i data-lucide="check-circle" class="w-3 h-3"></i> ' + stepEl.textContent.replace(/^.*?\s/, '');
          } else if (i === s.step) {
            stepEl.className = 'flex items-center gap-2 text-[#2563EB] font-semibold';
            stepEl.innerHTML = '<i data-lucide="loader-2" class="w-3 h-3 animate-spin"></i> ' + stepEl.textContent.replace(/^.*?\s/, '');
          } else {
            stepEl.className = 'flex items-center gap-2 text-[#98A2B3]';
            stepEl.innerHTML = '<i data-lucide="circle" class="w-3 h-3"></i> ' + stepEl.textContent.replace(/^.*?\s/, '');
          }
        }
        if (typeof lucide !== 'undefined') lucide.createIcons();
        idx++;
      }, 600);
    }
